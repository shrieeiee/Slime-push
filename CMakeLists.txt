cmake_minimum_required(VERSION 3.16)
project(SlimePush VERSION 1.2.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
# MSVC: bake the C/C++ runtime into the exe so no Visual C++ redistributable is needed
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(SLIMEPUSH_STATIC_SDL "Link SDL2 statically for a self-contained binary" OFF)
find_package(SDL2 REQUIRED)
find_package(Threads REQUIRED)

add_executable(SlimePush src/main.cpp src/game.cpp src/net.cpp src/gfx.cpp src/audio.cpp src/music.cpp)
if(MSVC)
  target_compile_options(SlimePush PRIVATE /W3 /wd4068 /wd4244 /wd4267 /wd4996 /utf-8)
  target_compile_definitions(SlimePush PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(SlimePush PRIVATE -Wall -Wextra -Wno-stringop-truncation)
endif()

if(SLIMEPUSH_STATIC_SDL AND TARGET SDL2::SDL2-static)
  set(SDL_LIB SDL2::SDL2-static)
else()
  set(SDL_LIB SDL2::SDL2)
endif()

if(WIN32)
  enable_language(RC)
  set_target_properties(SlimePush PROPERTIES WIN32_EXECUTABLE ON)
  target_sources(SlimePush PRIVATE win/slimepush.rc)
  if(MSVC)
    target_sources(SlimePush PRIVATE win/slimepush.manifest)
  endif()
  target_link_libraries(SlimePush PRIVATE SDL2::SDL2main ${SDL_LIB} ws2_32 iphlpapi)
  if(MINGW)
    target_link_options(SlimePush PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
else()
  target_link_libraries(SlimePush PRIVATE ${SDL_LIB} Threads::Threads)
  if(SLIMEPUSH_STATIC_SDL)
    target_link_options(SlimePush PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
